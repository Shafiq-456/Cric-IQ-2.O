    22→interface NotificationsState {
    23→  matchStartAlerts: boolean;
    24→  aiInsightDigest: boolean;
    25→  playerPerformanceAlerts: boolean;
    26→  weeklyAnalyticsReport: boolean;
    27→}
    28→
    29→interface SettingsState {
    30→  darkMode: boolean;
    31→  language: Language;
    32→  emailNotifications: boolean;
    33→  notifications: NotificationsState;
    34→}
    35→
    36→function loadSettings(): SettingsState {
    37→  if (typeof window === 'undefined') {
    38→    return {
    39→      darkMode: true,
    40→      language: 'English',
    41→      emailNotifications: true,
    42→      notifications: {
    43→        matchStartAlerts: true,
    44→        aiInsightDigest: true,
    45→        playerPerformanceAlerts: false,
    46→        weeklyAnalyticsReport: true,
    47→      },
    48→    };
    49→  }
    50→  try {
    51→    const raw = localStorage.getItem('criciq-settings');
    52→    if (raw) {
    53→      const parsed = JSON.parse(raw) as Partial<SettingsState>;
    54→      return {
    55→        darkMode: parsed.darkMode ?? true,
    56→        language: parsed.language ?? 'English',
    57→        emailNotifications: parsed.emailNotifications ?? true,
    58→        notifications: {
    59→          matchStartAlerts: parsed.notifications?.matchStartAlerts ?? true,
    60→          aiInsightDigest: parsed.notifications?.aiInsightDigest ?? true,
    61→          playerPerformanceAlerts: parsed.notifications?.playerPerformanceAlerts ?? false,
    62→          weeklyAnalyticsReport: parsed.notifications?.weeklyAnalyticsReport ?? true,
    63→        },
    64→      };
    65→    }
    66→  } catch {
    67→    // ignore parse errors
    68→  }
    69→  return {
    70→    darkMode: true,
    71→    language: 'English',
    72→    emailNotifications: true,
    73→    notifications: {
    74→      matchStartAlerts: true,
    75→      aiInsightDigest: true,
    76→      playerPerformanceAlerts: false,
    77→      weeklyAnalyticsReport: true,
    78→    },
    79→  };
    80→}
    81→
    82→function persistSettings(settings: SettingsState) {
    83→  if (typeof window !== 'undefined') {
    84→    localStorage.setItem('criciq-settings', JSON.stringify(settings));
    85→  }
    86→}

   529→                          Setup
   530→                        </Badge>
   531→                      )}
   532→                    </motion.div>
   533→                  </AnimatePresence>
   534→                </div>
   535→              </div>
   536→            </CardContent>
   537→          </Card>
   538→        </motion.div>
   539→      </div>
   540→
   541→      {/* ───── Save Button ───── */}
   542→      <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ delay: 0.4 }}>
   543→        <Button
   544→          className="w-full bg-[oklch(0.65_0.22_240)] hover:bg-[oklch(0.58_0.24_240)] text-white"
   545→          onClick={handleSave}
   546→          disabled={saving}
   547→        >
   548→          {saving ? (
   549→            <>
   550→              <Loader2 size={14} className="mr-2 animate-spin" /> Saving…
   551→            </>
   552→          ) : (
   553→            <>
   554→              <Save size={14} className="mr-2" /> Save Changes
   555→            </>
   556→          )}
   557→        </Button>
   558→      </motion.div>
   559→
   560→      {/* ───── Change Password Dialog ───── */}
   561→      <Dialog open={passwordDialogOpen} onOpenChange={(open) => {
   562→        setPasswordDialogOpen(open);
   563→        if (!open) {
   564→          setOldPassword('');
   565→          setNewPassword('');
   566→          setConfirmPassword('');
   567→          setPasswordError('');
   568→        }
   569→      }}>
   570→        <DialogContent className="glass sm:max-w-md">
   571→          <DialogHeader>
   572→            <DialogTitle>Change Password</DialogTitle>
   573→          </DialogHeader>
   574→          <div className="space-y-4 py-2">
   575→            <div className="space-y-1.5">
   576→              <Label className="text-xs">Current Password</Label>
   577→              <Input
   578→                type="password"
   579→                value={oldPassword}
   580→                onChange={(e) => setOldPassword(e.target.value)}
   581→                placeholder="Enter current password"
   582→                className="bg-muted/50 border-border/50 text-sm"
   583→              />
   584→            </div>
   585→            <div className="space-y-1.5">
   586→              <Label className="text-xs">New Password</Label>
   587→              <Input
   588→                type="password"
   589→                value={newPassword}
   590→                onChange={(e) => setNewPassword(e.target.value)}
   591→                placeholder="Enter new password"
   592→                className="bg-muted/50 border-border/50 text-sm"
   593→              />
   594→            </div>
   595→            <div className="space-y-1.5">
   596→              <Label className="text-xs">Confirm New Password</Label>
   597→              <Input
   598→                type="password"
   599→                value={confirmPassword}
   600→                onChange={(e) => setConfirmPassword(e.target.value)}

   602→                className="bg-muted/50 border-border/50 text-sm"
   603→              />
   604→            </div>
   605→            {passwordError && (
   606→              <motion.p
   607→                initial={{ opacity: 0, y: -4 }}
   608→                animate={{ opacity: 1, y: 0 }}
   609→                className="text-xs text-red-400"
   610→              >
   611→                {passwordError}
   612→              </motion.p>
   613→            )}
   614→          </div>
   615→          <DialogFooter>
   616→            <Button variant="outline" onClick={() => {
   617→              setPasswordDialogOpen(false);
   618→              setOldPassword('');
   619→              setNewPassword('');
   620→              setConfirmPassword('');
   621→              setPasswordError('');
   622→            }}>
   623→              Cancel
   624→            </Button>
   625→            <Button
   626→              className="bg-[oklch(0.65_0.22_240)] hover:bg-[oklch(0.58_0.24_240)] text-white"
   627→              onClick={handlePasswordChange}
   628→            >
   629→              Change
   630→            </Button>
   631→          </DialogFooter>
   632→        </DialogContent>
   633→      </Dialog>
   634→    </div>
   635→  );
