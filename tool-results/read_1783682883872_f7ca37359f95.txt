    22→    22→    22→/* ─── 3D Cricket Ball with Mouse-Follow Tilt ─── */
    23→    23→    23→function CricketBall3D({ size = 280, className = '' }: { size?: number; className?: string }) {
    24→    24→    24→  const ref = useRef<HTMLDivElement>(null);
    25→    25→    25→  const mouseX = useMotionValue(0);
    26→    26→    26→  const mouseY = useMotionValue(0);
    27→    27→    27→  const springConfig = { stiffness: 80, damping: 20, mass: 0.5 };
    28→    28→    28→  const rotateX = useSpring(0, springConfig);
    29→    29→    29→  const rotateY = useSpring(0, springConfig);
    30→    30→    30→
    31→    31→    31→  useEffect(() => {
    32→    32→    32→    const el = ref.current;
    33→    33→    33→    if (!el) return;
    34→    34→    34→    const parent = el.closest('section') || window;
    35→    35→    35→
    36→    36→    36→    const handleMove = (e: MouseEvent) => {
    37→    37→    37→      const rect = el.getBoundingClientRect();
    38→    38→    38→      const cx = rect.left + rect.width / 2;
    39→    39→    39→      const cy = rect.top + rect.height / 2;
    40→    40→    40→      const dx = (e.clientX - cx) / (rect.width / 2);
    41→    41→    41→      const dy = (e.clientY - cy) / (rect.height / 2);
    42→    42→    42→      rotateX.set(-dy * 25);
    43→    43→    43→      rotateY.set(dx * 25);
    44→    44→    44→    };
    45→    45→    45→
    46→    46→    46→    const handleLeave = () => {
    47→    47→    47→      rotateX.set(0);
    48→    48→    48→      rotateY.set(0);
    49→    49→    49→    };
    50→    50→    50→
    51→    51→    51→    window.addEventListener('mousemove', handleMove);
    52→    52→    52→    window.addEventListener('mouseleave', handleLeave);
    53→    53→    53→    return () => {
    54→    54→    54→      window.removeEventListener('mousemove', handleMove);
    55→    55→    55→      window.removeEventListener('mouseleave', handleLeave);
    56→    56→    56→    };
    57→    57→    57→  }, [rotateX, rotateY]);
    58→    58→    58→
    59→    59→    59→  return (
    60→    60→    60→    <div ref={ref} className={`perspective-dramatic ${className}`}>
    61→    61→    61→      <motion.div
    62→    62→    62→        className="preserve-3d relative"
    63→    63→    63→        style={{ rotateX, rotateY, width: size, height: size }}
    64→    64→    64→      >
    65→    65→    65→        {/* Ball body */}
    66→    66→    66→        <div
    67→    67→    67→          className="w-full h-full rounded-full relative overflow-hidden"
    68→    68→    68→          style={{
    69→    69→    69→            boxShadow: `
    70→    70→    70→              0 0 ${size * 0.4}px rgba(192,57,43,0.18),
    71→    71→    71→              0 ${size * 0.12}px ${size * 0.35}px rgba(0,0,0,0.5),
    72→    72→    72→              0 0 ${size * 0.8}px rgba(192,57,43,0.08)
    73→    73→    73→            `,
    74→    74→    74→          }}
    75→    75→    75→        >
    76→    76→    76→          <img src="/cricket-ball.png" alt="" className="w-full h-full object-cover" />
    77→    77→    77→        </div>
    78→    78→    78→
    79→    79→    79→        {/* Shadow beneath ball */}
    80→    80→    80→        <div
    81→    81→    81→          className="absolute -bottom-6 left-1/2 -translate-x-1/2 w-3/4 h-4 rounded-full"
    82→    82→    82→          style={{
    83→    83→    83→            background: 'radial-gradient(ellipse, rgba(0,0,0,0.4) 0%, transparent 70%)',
    84→    84→    84→            filter: 'blur(8px)',
    85→    85→    85→          }}
    86→    86→    86→        />
    87→    87→    87→      </motion.div>
    88→    88→    88→    </div>
    89→    89→    89→  );
    90→    90→    90→}
