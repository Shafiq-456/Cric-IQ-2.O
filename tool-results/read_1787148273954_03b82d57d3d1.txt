    36→    36→function loadSettings(): SettingsState {
    37→    37→  if (typeof window === 'undefined') {
    38→    38→    return {
    39→    39→      darkMode: true,
    40→    40→      language: 'English',
    41→    41→      emailNotifications: true,
    42→    42→      notifications: {
    43→    43→        matchStartAlerts: true,
    44→    44→        aiInsightDigest: true,
    45→    45→        playerPerformanceAlerts: false,
    46→    46→        weeklyAnalyticsReport: true,
    47→    47→      },
    48→    48→    };
    49→    49→  }
    50→    50→  try {
    51→    51→    const raw = localStorage.getItem('criciq-settings');
    52→    52→    if (raw) {
    53→    53→      const parsed = JSON.parse(raw) as Partial<SettingsState>;
    54→    54→      return {
    55→    55→        darkMode: parsed.darkMode ?? true,
    56→    56→        language: parsed.language ?? 'English',
    57→    57→        emailNotifications: parsed.emailNotifications ?? true,
    58→    58→        notifications: {
    59→    59→          matchStartAlerts: parsed.notifications?.matchStartAlerts ?? true,
    60→    60→          aiInsightDigest: parsed.notifications?.aiInsightDigest ?? true,
    61→    61→          playerPerformanceAlerts: parsed.notifications?.playerPerformanceAlerts ?? false,
    62→    62→          weeklyAnalyticsReport: parsed.notifications?.weeklyAnalyticsReport ?? true,
    63→    63→        },
    64→    64→      };
    65→    65→    }
    66→    66→  } catch {
    67→    67→    // ignore parse errors
    68→    68→  }
    69→    69→  return {
    70→    70→    darkMode: true,
    71→    71→    language: 'English',
    72→    72→    emailNotifications: true,
    73→    73→    notifications: {
    74→    74→      matchStartAlerts: true,
