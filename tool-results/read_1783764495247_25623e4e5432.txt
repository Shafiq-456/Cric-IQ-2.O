    35→/* ═══ Scroll-linked 3D Section Wrapper ═══ */
    36→function Section3D({ children, className = '', id = '' }: { children: React.ReactNode; className?: string; id?: string }) {
    37→  const ref = useRef<HTMLElement>(null);
    38→  const inView = useInView(ref, { once: true, margin: '-80px' });
    39→  const { scrollYProgress } = useScroll({ target: ref, offset: ['start end', 'end start'] });
    40→  const rotateX = useTransform(scrollYProgress, [0, 0.5, 1], [4, 0, -2]);
    41→  const y = useTransform(scrollYProgress, [0, 1], [30, -20]);
    42→
    43→  return (
    44→    <motion.section
    45→      ref={ref}
    46→      id={id}
    47→      className={`perspective-subtle ${className}`}
    48→      style={{ rotateX, y }}
    49→    >
    50→      {typeof children === 'function' ? children({ inView }) : children}
    51→    </motion.section>
    52→  );
    53→}

    55→/* ═══ 3D Card with hover tilt ═══ */
    56→function Card3D({ children, className = '', glowColor = 'oklch(0.65 0.20 155)' }: { children: React.ReactNode; className?: string; glowColor?: string }) {
    57→  const ref = useRef<HTMLDivElement>(null);
    58→
    59→  const handleMouseMove = (e: React.MouseEvent) => {
    60→    const el = ref.current;
    61→    if (!el) return;
    62→    const rect = el.getBoundingClientRect();
    63→    const x = (e.clientX - rect.left) / rect.width - 0.5;
    64→    const y = (e.clientY - rect.top) / rect.height - 0.5;
    65→    el.style.transform = `perspective(800px) rotateY(${x * 12}deg) rotateX(${-y * 12}deg) scale(1.02)`;
    66→    el.style.boxShadow = `${-x * 20}px ${y * 20}px 60px rgba(0,0,0,0.3), 0 0 40px ${glowColor}15`;
    67→  };
    68→
    69→  const handleMouseLeave = () => {
    70→    const el = ref.current;
    71→    if (!el) return;
    72→    el.style.transform = 'perspective(800px) rotateY(0) rotateX(0) scale(1)';
    73→    el.style.boxShadow = '';
    74→  };
    75→
    76→  return (
    77→    <div
    78→      ref={ref}
    79→      className="card-3d"
    80→      onMouseMove={handleMouseMove}
    81→      onMouseLeave={handleMouseLeave}
    82→      style={{ transition: 'transform 0.4s cubic-bezier(0.16, 1, 0.3, 1), box-shadow 0.4s ease' }}
    83→    >
    84→      {children}
    85→    </div>
    86→  );
    87→}
