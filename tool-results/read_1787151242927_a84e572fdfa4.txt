    23→interface NotificationsState {
    24→  matchStartAlerts: boolean;
    25→  aiInsightDigest: boolean;
    26→  playerPerformanceAlerts: boolean;
    27→  weeklyAnalyticsReport: boolean;
    28→}

    30→interface SettingsState {
    31→  darkMode: boolean;
    32→  language: Language;
    33→  emailNotifications: boolean;
    34→  notifications: NotificationsState;
    35→}
    36→
    37→function loadSettings(): SettingsState {
    38→  if (typeof window === 'undefined') {
    39→    return {
    40→      darkMode: true,
    41→      language: 'English',
    42→      emailNotifications: true,
    43→      notifications: {
    44→        matchStartAlerts: true,
    45→        aiInsightDigest: true,
    46→        playerPerformanceAlerts: false,
    47→        weeklyAnalyticsReport: true,
    48→      },
    49→    };
    50→  }
    51→  try {
    52→    const raw = localStorage.getItem('criciq-settings');
    53→    if (raw) {
    54→      const parsed = JSON.parse(raw) as Partial<SettingsState>;
    55→      return {
    56→        darkMode: parsed.darkMode ?? true,
    57→        language: parsed.language ?? 'English',
    58→        emailNotifications: parsed.emailNotifications ?? true,
    59→        notifications: {
    60→          matchStartAlerts: parsed.notifications?.matchStartAlerts ?? true,
    61→          aiInsightDigest: parsed.notifications?.aiInsightDigest ?? true,
    62→          playerPerformanceAlerts: parsed.notifications?.playerPerformanceAlerts ?? false,
    63→          weeklyAnalyticsReport: parsed.notifications?.weeklyAnalyticsReport ?? true,
    64→        },
    65→      };
    66→    }
    67→  } catch {
    68→    // ignore parse errors
    69→  }
    70→  return {
    71→    darkMode: true,
    72→    language: 'English',
    73→    emailNotifications: true,
    74→    notifications: {
    75→      matchStartAlerts: true,
    76→      aiInsightDigest: true,
    77→      playerPerformanceAlerts: false,
    78→      weeklyAnalyticsReport: true,
    79→    },
    80→  };
    81→}
    82→
    83→function persistSettings(settings: SettingsState) {
    84→  if (typeof window !== 'undefined') {
    85→    localStorage.setItem('criciq-settings', JSON.stringify(settings));
    86→  }
    87→}

   530→                        <Badge variant="outline" className="text-[10px] border-[oklch(0.78_0.16_85/30)] text-[oklch(0.78_0.16_85)]">
   531→                          Setup
   532→                        </Badge>
   533→                      )}
   534→                    </motion.div>
   535→                  </AnimatePresence>
   536→                </div>
   537→              </div>
   538→            </CardContent>
   539→          </Card>
   540→        </motion.div>
   541→      </div>
   542→
   543→      {/* ───── Save Button ───── */}
   544→      <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ delay: 0.4 }}>
   545→        <Button
   546→          className="w-full bg-[oklch(0.65_0.22_240)] hover:bg-[oklch(0.58_0.24_240)] text-white"
   547→          onClick={handleSave}
   548→          disabled={saving}
   549→        >
   550→          {saving ? (
   551→            <>
   552→              <Loader2 size={14} className="mr-2 animate-spin" /> Saving…
   553→            </>
   554→          ) : (
   555→            <>
   556→              <Save size={14} className="mr-2" /> Save Changes
   557→            </>
   558→          )}
   559→        </Button>
   560→      </motion.div>
   561→
   562→      {/* ───── Change Password Dialog ───── */}
   563→      <Dialog open={passwordDialogOpen} onOpenChange={(open) => {
   564→        setPasswordDialogOpen(open);
   565→        if (!open) {
   566→          setOldPassword('');
   567→          setNewPassword('');
   568→          setConfirmPassword('');
   569→          setPasswordError('');
   570→        }
   571→      }}>
   572→        <DialogContent className="glass sm:max-w-md">
   573→          <DialogHeader>
   574→            <DialogTitle>Change Password</DialogTitle>
   575→          </DialogHeader>
   576→          <div className="space-y-4 py-2">
   577→            <div className="space-y-1.5">
   578→              <Label className="text-xs">Current Password</Label>
   579→              <Input
   580→                type="password"
   581→                value={oldPassword}
   582→                onChange={(e) => setOldPassword(e.target.value)}
   583→                placeholder="Enter current password"
   584→                className="bg-muted/50 border-border/50 text-sm"
   585→              />
   586→            </div>
   587→            <div className="space-y-1.5">
   588→              <Label className="text-xs">New Password</Label>
   589→              <Input
   590→                type="password"
   591→                value={newPassword}
   592→                onChange={(e) => setNewPassword(e.target.value)}
   593→                placeholder="Enter new password"

   610→                animate={{ opacity: 1, y: 0 }}

   612→              >
   613→                {passwordError}

   615→            )}
   616→          </div>
   617→          <DialogFooter>
   618→            <Button variant="outline" onClick={() => {
   619→              setPasswordDialogOpen(false);
   620→              setOldPassword('');

   622→              setConfirmPassword('');
   623→              setPasswordError('');
   624→            }}>
   625→              Cancel
   626→            </Button>
   627→            <Button
   628→              className="bg-[oklch(0.65_0.22_240)] hover:bg-[oklch(0.58_0.24_240)] text-white"
   629→              onClick={handlePasswordChange}
   630→            >
   631→              Change
   632→            </Button>
   633→          </DialogFooter>
   634→        </DialogContent>
   635→      </Dialog>
   636→    </div>
   637→  );
