    37→    37→    37→function Section3D({ children, className = '', id = '' }: { children: React.ReactNode; className?: string; id?: string }) {
    38→    38→    38→  const ref = useRef<HTMLElement>(null);
    39→    39→    39→  const inView = useInView(ref, { once: true, margin: '-80px' });
    40→    40→    40→  const { scrollYProgress } = useScroll({ target: ref, offset: ['start end', 'end start'] });
    41→    41→    41→  const rotateX = useTransform(scrollYProgress, [0, 0.5, 1], [4, 0, -2]);
    42→    42→    42→  const y = useTransform(scrollYProgress, [0, 1], [30, -20]);
    43→    43→    43→
    44→    44→    44→  return (
    45→    45→    45→    <motion.section
    46→    46→    46→      ref={ref}
    47→    47→    47→      id={id}
    48→    48→    48→      className={`perspective-subtle ${className}`}
    49→    49→    49→      style={{ rotateX, y }}
    50→    50→    50→    >
    51→    51→    51→      {typeof children === 'function' ? children({ inView }) : children}
    52→    52→    52→    </motion.section>
    53→    53→    53→  );
    54→    54→    54→}

    57→    57→    57→function Card3D({ children, className = '', glowColor = 'oklch(0.65 0.20 155)' }: { children: React.ReactNode; className?: string; glowColor?: string }) {
    58→    58→    58→  const ref = useRef<HTMLDivElement>(null);
    59→    59→    59→
    60→    60→    60→  const handleMouseMove = (e: React.MouseEvent) => {
    61→    61→    61→    const el = ref.current;
    62→    62→    62→    if (!el) return;
    63→    63→    63→    const rect = el.getBoundingClientRect();
    64→    64→    64→    const x = (e.clientX - rect.left) / rect.width - 0.5;
    65→    65→    65→    const y = (e.clientY - rect.top) / rect.height - 0.5;
    66→    66→    66→    el.style.transform = `perspective(800px) rotateY(${x * 12}deg) rotateX(${-y * 12}deg) scale(1.02)`;
    67→    67→    67→    el.style.boxShadow = `${-x * 20}px ${y * 20}px 60px rgba(0,0,0,0.3), 0 0 40px ${glowColor}15`;
    68→    68→    68→  };
    69→    69→    69→
    70→    70→    70→  const handleMouseLeave = () => {
    71→    71→    71→    const el = ref.current;
    72→    72→    72→    if (!el) return;
    73→    73→    73→    el.style.transform = 'perspective(800px) rotateY(0) rotateX(0) scale(1)';
    74→    74→    74→    el.style.boxShadow = '';
    75→    75→    75→  };
    76→    76→    76→
    77→    77→    77→  return (
    78→    78→    78→    <div
    79→    79→    79→      ref={ref}
    80→    80→    80→      className="card-3d"
    81→    81→    81→      onMouseMove={handleMouseMove}
    82→    82→    82→      onMouseLeave={handleMouseLeave}
    83→    83→    83→      style={{ transition: 'transform 0.4s cubic-bezier(0.16, 1, 0.3, 1), box-shadow 0.4s ease' }}
    84→    84→    84→    >
    85→    85→    85→      {children}
    86→    86→    86→    </div>
    87→    87→    87→  );
    88→    88→    88→}
