    22→/* ─── 3D Cricket Ball with Mouse-Follow Tilt ─── */
    23→function CricketBall3D({ size = 280, className = '' }: { size?: number; className?: string }) {
    24→  const ref = useRef<HTMLDivElement>(null);
    25→  const mouseX = useMotionValue(0);
    26→  const mouseY = useMotionValue(0);
    27→  const springConfig = { stiffness: 80, damping: 20, mass: 0.5 };
    28→  const rotateX = useSpring(0, springConfig);
    29→  const rotateY = useSpring(0, springConfig);
    30→
    31→  useEffect(() => {
    32→    const el = ref.current;
    33→    if (!el) return;
    34→    const parent = el.closest('section') || window;
    35→
    36→    const handleMove = (e: MouseEvent) => {
    37→      const rect = el.getBoundingClientRect();
    38→      const cx = rect.left + rect.width / 2;
    39→      const cy = rect.top + rect.height / 2;
    40→      const dx = (e.clientX - cx) / (rect.width / 2);
    41→      const dy = (e.clientY - cy) / (rect.height / 2);
    42→      rotateX.set(-dy * 25);
    43→      rotateY.set(dx * 25);
    44→    };
    45→
    46→    const handleLeave = () => {
    47→      rotateX.set(0);
    48→      rotateY.set(0);
    49→    };
    50→
    51→    window.addEventListener('mousemove', handleMove);
    52→    window.addEventListener('mouseleave', handleLeave);
    53→    return () => {
    54→      window.removeEventListener('mousemove', handleMove);
    55→      window.removeEventListener('mouseleave', handleLeave);
    56→    };
    57→  }, [rotateX, rotateY]);
    58→
    59→  return (
    60→    <div ref={ref} className={`perspective-dramatic ${className}`}>
    61→      <motion.div
    62→        className="preserve-3d relative"
    63→        style={{ rotateX, rotateY, width: size, height: size }}
    64→      >
    65→        {/* Ball body */}
    66→        <div
    67→          className="w-full h-full rounded-full relative"
    68→          style={{
    69→            background: `radial-gradient(circle at 32% 26%, #f07058 0%, #d44535 22%, #b02a1c 45%, #7a1a10 75%, #4a0e08 100%)`,
    70→            boxShadow: `
    71→              inset -${size * 0.06}px -${size * 0.06}px ${size * 0.15}px rgba(0,0,0,0.7),
    72→              inset ${size * 0.04}px ${size * 0.04}px ${size * 0.1}px rgba(255,220,200,0.1),
    73→              0 0 ${size * 0.4}px rgba(192,57,43,0.18),
    74→              0 ${size * 0.12}px ${size * 0.35}px rgba(0,0,0,0.5),
    75→              0 0 ${size * 0.8}px rgba(192,57,43,0.08)
    76→            `,
    77→          }}
    78→        >
    79→          {/* Seam stitching — primary */}
    80→          <svg className="absolute inset-0 w-full h-full" viewBox="0 0 200 200">
    81→            <defs>
    82→              <filter id="seam-glow">
    83→                <feGaussianBlur stdDeviation="1.5" result="blur" />
    84→                <feMerge>
    85→                  <feMergeNode in="blur" />
    86→                  <feMergeNode in="SourceGraphic" />
    87→                </feMerge>
    88→              </filter>
    89→            </defs>
    90→            {/* Main seam curve */}
    91→            <path
    92→              d="M100 18 Q112 100 100 182"
    93→              stroke="rgba(255,255,255,0.5)"
    94→              strokeWidth="2.5"
    95→              fill="none"
    96→              strokeDasharray="6 4"
    97→              filter="url(#seam-glow)"
    98→            />
    99→            {/* Secondary seam curve */}
   100→            <path
   101→              d="M92 18 Q80 100 92 182"
   102→              stroke="rgba(255,255,255,0.3)"
   103→              strokeWidth="1.5"
   104→              fill="none"
   105→              strokeDasharray="6 4"
   106→            />
   107→            {/* Equator line */}
   108→            <line x1="6" y1="100" x2="194" y2="100" stroke="rgba(255,255,255,0.2)" strokeWidth="1.5" />
   109→            {/* Stitch marks along seam */}
   110→            {Array.from({ length: 16 }).map((_, i) => {
   111→              const y = 30 + i * 10;
   112→              const x1 = 100 + Math.sin((y - 100) / 100 * Math.PI) * 8;
   113→              const x2 = 100 + Math.sin((y - 100) / 100 * Math.PI) * 14;
   114→              return (
   115→                <line
   116→                  key={i}
   117→                  x1={x1} y1={y} x2={x2} y2={y}
   118→                  stroke="rgba(255,255,255,0.18)"

   123→            {/* Specular highlight */}
