    40→  const { scrollYProgress } = useScroll({ target: ref, offset: ['start end', 'end start'] });
    41→  const rotateX = useTransform(scrollYProgress, [0, 0.5, 1], [4, 0, -2]);
    42→  const y = useTransform(scrollYProgress, [0, 1], [30, -20]);
    43→
    44→  return (
    45→    <motion.section
    46→      ref={ref}
    47→      id={id}
    48→      className={`perspective-subtle ${className}`}
    49→      style={{ rotateX, y }}
    50→    >
    51→      {typeof children === 'function' ? children({ inView }) : children}
    52→    </motion.section>
    53→  );
    54→}

    56→/* ═══ 3D Card with hover tilt ═══ */
    57→function Card3D({ children, className = '', glowColor = 'oklch(0.65 0.20 155)' }: { children: React.ReactNode; className?: string; glowColor?: string }) {
    58→  const ref = useRef<HTMLDivElement>(null);
    59→
    60→  const handleMouseMove = (e: React.MouseEvent) => {
    61→    const el = ref.current;
    62→    if (!el) return;
    63→    const rect = el.getBoundingClientRect();
    64→    const x = (e.clientX - rect.left) / rect.width - 0.5;
    65→    const y = (e.clientY - rect.top) / rect.height - 0.5;
    66→    el.style.transform = `perspective(800px) rotateY(${x * 12}deg) rotateX(${-y * 12}deg) scale(1.02)`;
    67→    el.style.boxShadow = `${-x * 20}px ${y * 20}px 60px rgba(0,0,0,0.3), 0 0 40px ${glowColor}15`;
    68→  };
    69→
    70→  const handleMouseLeave = () => {
    71→    const el = ref.current;
    72→    if (!el) return;
    73→    el.style.transform = 'perspective(800px) rotateY(0) rotateX(0) scale(1)';
    74→    el.style.boxShadow = '';
    75→  };
    76→
    77→  return (
    78→    <div
    79→      ref={ref}
    80→      className="card-3d"
    81→      onMouseMove={handleMouseMove}
    82→      onMouseLeave={handleMouseLeave}
    83→      style={{ transition: 'transform 0.4s cubic-bezier(0.16, 1, 0.3, 1), box-shadow 0.4s ease' }}
    84→    >
    85→      {children}
    86→    </div>
    87→  );
    88→}
