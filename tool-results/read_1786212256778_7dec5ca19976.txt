    27→function generateSmartResponse(query: string): ChatMessage {
    28→  const q = query.toLowerCase();
    29→
    30→  // 1. COMPARISON queries: "compare X vs Y", "X vs Y"
    31→  const compareMatch = q.match(/compare\s+(.+?)\s+vs\.?\s+(.+)/) || q.match(/(.+?)\s+vs\.?\s+(.+)/);
    32→  if (compareMatch) {
    33→    const nameA = compareMatch[1].replace(/\bin\s+away\s+tests\b/i, '').replace(/['']/g, '').trim();
    34→    const nameB = compareMatch[2].replace(/\bin\s+away\s+tests\b/i, '').replace(/['']/g, '').trim();
    35→    const playerA = findPlayer(nameA);
    36→    const playerB = findPlayer(nameB);
    37→
    38→    if (playerA && playerB) {
    39→      return buildComparisonResponse(playerA, playerB, query);
    40→    }
    41→    if (playerA && !playerB) {
    42→      return buildSinglePlayerResponse(playerA, query);
    43→    }
    44→    if (!playerA && playerB) {
    45→      return buildSinglePlayerResponse(playerB, query);
    46→    }
    47→  }
    48→
    49→  // 2. BOWLING stats queries
    50→  if (q.includes('bowling') || q.includes('wicket') || q.includes('economy') || q.includes('best bowling figures')) {
    51→    return buildBowlingResponse(q, query);
    52→  }
    53→
    54→  // 3. BATTING stats queries
    55→  if (q.includes('batting') || q.includes('runs') || q.includes('average') || q.includes('strike rate') || q.includes('century') || q.includes('scores')) {
    56→    return buildBattingResponse(q, query);
    57→  }
    58→
    59→  // 4. Single player lookup
    60→  const foundPlayer = players.find(p => q.includes(p.name.toLowerCase()));
    61→  if (foundPlayer) {
    62→    return buildSinglePlayerResponse(foundPlayer, query);
    63→  }
    64→
    65→  // 5. MATCH / TEAM lookup
    66→  const foundMatch = matches.find(m =>
    67→    q.includes(m.teamA.toLowerCase()) && q.includes(m.teamB.toLowerCase())
    68→  );
    69→  if (foundMatch) {
    70→    return buildMatchResponse(foundMatch, query);
    71→  }
    72→
    73→  // Check by single team
    74→  const foundTeamMatches = matches.filter(m =>
    75→    q.includes(m.teamA.toLowerCase()) || q.includes(m.teamB.toLowerCase())
    76→  );
    77→  if (foundTeamMatches.length > 0) {
    78→    const teamName = foundTeamMatches[0].teamA.toLowerCase().includes(q) || q.includes(foundTeamMatches[0].teamA.toLowerCase())
    79→      ? foundTeamMatches[0].teamA
    80→      : foundTeamMatches[0].teamB;
    81→    return buildTeamMatchesResponse(teamName, foundTeamMatches, query);
    82→  }
    83→
    84→  // 6. TOP / BEST / RANKING queries
    85→  if (q.includes('top') || q.includes('best') || q.includes('ranking') || q.includes('icc')) {
    86→    return buildRankingResponse(q, query);
    87→  }
    88→
    89→  // 7. UPCOMING / LIVE / SCHEDULE queries
    90→  if (q.includes('upcoming') || q.includes('schedule') || q.includes('next match') || q.includes('live')) {
    91→    return buildScheduleResponse(q, query);
    92→  }
    93→
    94→  // 8. Under-25 / young bowlers etc.
    95→  if (q.includes('under 25') || q.includes('young') || q.includes('fast bowler')) {
    96→    return buildYoungPlayersResponse(q, query);
    97→  }
    98→
    99→  // 9. DRS / head-to-head / specific topic without direct match
   100→  if (q.includes('drs') || q.includes('head-to-head') || q.includes('head to head')) {
   101→    return buildTopicResponse(q, query);
   102→  }
   103→
   104→  // 10. Fallback: unrecognized
   105→  return buildFallbackResponse(query);
   106→}

   108→function findPlayer(name: string): Player | undefined {
   109→  const clean = name.replace(/['']/g, '').trim().toLowerCase();
   110→  return players.find(p => {
   111→    const pName = p.name.toLowerCase();
   112→    if (pName === clean) return true;
   113→    // Check if all significant words in `clean` appear in the player name
   114→    const words = clean.split(/\s+/).filter(w => w.length > 2);
   115→    return words.every(w => pName.includes(w));
   116→  });
   117→}

   119→function buildComparisonResponse(a: Player, b: Player, query: string): ChatMessage {
   120→  const runsDiff = a.runs - b.runs;
   121→  const avgDiff = (a.average - b.average).toFixed(2);
   122→  const betterRuns = runsDiff > 0 ? a.name : b.name;
   123→  const betterAvg = a.average > b.average ? a.name : b.name;
   124→  const isBatting = a.role === 'Batsman' || b.role === 'Batsman' || a.role === 'All-rounder' || b.role === 'All-rounder';
   125→  const isBowling = a.role === 'Bowler' || b.role === 'Bowler' || a.role === 'All-rounder' || b.role === 'All-rounder';
   126→
   127→  let content = `Here's a detailed comparison between **${a.name}** (${a.flag} ${a.country}) and **${b.name}** (${b.flag} ${b.country}):\n\n`;
   128→
   129→  if (isBatting) {
   130→    content += `**Batting Comparison:**\n`;
   131→    content += `• **${a.name}**: ${a.runs.toLocaleString()} runs in ${a.matches} matches at an average of **${a.average}** (SR: ${a.strikeRate})\n`;
   132→    content += `• **${b.name}**: ${b.runs.toLocaleString()} runs in ${b.matches} matches at an average of **${b.average}** (SR: ${b.strikeRate})\n\n`;
   133→    content += `${betterRuns} has scored more total runs, while ${betterAvg} holds the higher batting average. `;
   134→  }
   135→
   136→  if (isBowling) {
   137→    content += `**Bowling Comparison:**\n`;
   138→    content += `• **${a.name}**: ${a.wickets} wickets in ${a.matches} matches at an average of **${a.average}**, economy ${a.economy}\n`;
   139→    content += `• **${b.name}**: ${b.wickets} wickets in ${b.matches} matches at an average of **${b.average}**, economy ${b.economy}\n\n`;

   145→
   146→  return {
   147→    id: `msg-${Date.now() + 1}`,
   148→    role: 'assistant',
   149→    content,
   150→    citations: [
   151→      { source: 'ICC Player Rankings 2026', snippet: `${a.name}: #${a.iccRanking} | ${b.name}: #${b.iccRanking}` },
   152→      { source: 'ESPNcricinfo Statsguru', snippet: `Career records comparison retrieved` },

   154→    confidence: 92,
   155→    timestamp: new Date(),
   156→    stats: [
   157→      { label: `${a.name} Runs`, value: a.runs.toLocaleString() },
   158→      { label: `${a.name} Avg`, value: a.average.toString() },
   159→      { label: `${b.name} Runs`, value: b.runs.toLocaleString() },

   162→      { label: `${b.name} Rating`, value: `${b.rating}/10` },

   277→function buildMatchResponse(match: Match, query: string): ChatMessage {
   278→  let content = `**${match.teamA} ${match.flagA} vs ${match.teamB} ${match.flagB}**\n\n`;
   279→  content += `• **Competition**: ${match.competition}\n`;
   280→  content += `• **Date**: ${match.date}\n`;
   281→  content += `• **Venue**: ${match.venue}\n`;
   282→  content += `• **Status**: **${match.status.toUpperCase()}**\n`;
   283→
   284→  if (match.scoreA) content += `• **${match.teamA} Score**: ${match.scoreA}\n`;
   285→  if (match.scoreB) content += `• **${match.teamB} Score**: ${match.scoreB}\n`;
   286→  if (match.result) content += `• **Result**: ${match.result}\n`;
   287→  if (match.motm) content += `• **Man of the Match**: ${match.motm}\n`;
   288→
   289→  if (match.highlights.length > 0) {
   290→    content += `\n**Highlights:**\n`;
   291→    match.highlights.forEach(h => { content += `• ${h}\n`; });
   292→  }
   293→
   294→  const playerMention = match.motm ? findPlayer(match.motm) : null;
   295→  const citations = [
   296→    { source: `${match.competition} Match Center`, snippet: `${match.teamA} vs ${match.teamB} — ${match.date}` },
   297→  ];
   298→  if (playerMention) {
   299→    citations.push({ source: 'Player Performance Data', snippet: `${playerMention.name}: ${playerMention.recentScores.join(', ')}` });
   300→  }
   301→
   302→  return {
   303→    id: `msg-${Date.now() + 1}`,
   304→    role: 'assistant',
   305→    content,
   306→    citations,
   307→    confidence: match.status === 'completed' ? 96 : match.status === 'live' ? 93 : 85,
   308→    timestamp: new Date(),
   309→    stats: [
   310→      { label: 'Competition', value: match.competition.split(' ').slice(-2).join(' ') },
   311→      { label: 'Status', value: match.status },
   312→      ...(match.scoreA ? [{ label: `${match.teamA}`, value: match.scoreA }] : []),
   313→      ...(match.scoreB ? [{ label: `${match.teamB}`, value: match.scoreB }] : []),
   314→      ...(match.motm ? [{ label: 'MoTM', value: match.motm }] : []),
   315→    ],
   316→  };
   317→}

   367→function buildRankingResponse(q: string, query: string): ChatMessage {
   368→  const isBowlingRank = q.includes('bowler') || q.includes('wicket') || q.includes('bowling');
   369→  const isBattingRank = q.includes('batsman') || q.includes('batting') || q.includes('run');
   370→
   371→  let sorted: Player[];
   372→  let statLabel: string;
   373→
   374→  if (isBowlingRank) {
   375→    sorted = players.filter(p => p.wickets > 0).sort((a, b) => a.iccRanking - b.iccRanking);
   376→    statLabel = 'Wickets';
   377→  } else {
   378→    sorted = players.filter(p => p.runs > 0).sort((a, b) => a.iccRanking - b.iccRanking);
   379→    statLabel = 'Runs';
   380→  }
   381→
   382→  const content = `**ICC Rankings** (from our database):\n\n` +
   383→    sorted.slice(0, 5).map((p, i) => {
   384→      const stat = isBowlingRank ? `${p.wickets} wickets, econ ${p.economy}` : `${p.runs.toLocaleString()} runs, avg ${p.average}`;
   385→      return `• **#${p.iccRanking}** ${p.name} ${p.flag} — ${stat} (rating: ${p.rating}/10)`;
   386→    }).join('\n') + '\n\n' +
   387→    `Note: Rankings are based on the ICC official ranking system as available in our data.`;
   388→
   389→  return {
   390→    id: `msg-${Date.now() + 1}`,
   391→    role: 'assistant',
   392→    content,
   393→    citations: [
   394→      { source: 'ICC Official Rankings 2026', snippet: `Top 5 ${isBowlingRank ? 'bowlers' : 'batsmen'} by ICC ranking` },
   395→    ],
   396→    confidence: 94,
   397→    timestamp: new Date(),
   398→    stats: sorted.slice(0, 5).map(p => ({
   399→      label: `#${p.iccRanking}`,
   400→      value: p.name,
   401→    })),
   402→  };
   403→}

   405→function buildScheduleResponse(q: string, query: string): ChatMessage {
   406→  const upcoming = matches.filter(m => m.status === 'upcoming');
   407→  const live = matches.filter(m => m.status === 'live');
   408→
   409→  let content = '';

   411→    content += `🔴 **Live Matches:**\n`;

   413→      content += `• **${m.teamA} ${m.flagA} vs ${m.flagB} ${m.teamB}** — ${m.scoreA || ''} / ${m.scoreB || ''} at ${m.venue} (${m.competition})\n`;
   414→    });
   415→    content += '\n';
   416→  }
   417→
   418→  if (upcoming.length > 0) {
   419→    content += `📅 **Upcoming Matches:**\n`;
   420→    upcoming.forEach(m => {
   421→      content += `• **${m.teamA} ${m.flagA} vs ${m.flagB} ${m.teamB}** — ${m.date} at ${m.venue} (${m.competition})\n`;
   422→      if (m.highlights.length > 0) content += `  _${m.highlights.join(', ')}_\n`;
   423→    });
   424→  }
   425→
   426→  if (!content) {
   427→    content = 'No upcoming or live matches found in the current data.';
