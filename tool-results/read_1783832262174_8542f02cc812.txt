    22→/* ─── 3D Cricket Ball with Mouse-Follow Tilt ─── */
    23→function CricketBall3D({ size = 280, className = '' }: { size?: number; className?: string }) {
    24→  const ref = useRef<HTMLDivElement>(null);
    25→  const mouseX = useMotionValue(0);
    26→  const mouseY = useMotionValue(0);
    27→  const springConfig = { stiffness: 80, damping: 20, mass: 0.5 };
    28→  const rotateX = useSpring(0, springConfig);
    29→  const rotateY = useSpring(0, springConfig);
    30→
    31→  useEffect(() => {
    32→    const el = ref.current;
    33→    if (!el) return;
    34→    const parent = el.closest('section') || window;
    35→
    36→    const handleMove = (e: MouseEvent) => {
    37→      const rect = el.getBoundingClientRect();
    38→      const cx = rect.left + rect.width / 2;
    39→      const cy = rect.top + rect.height / 2;
    40→      const dx = (e.clientX - cx) / (rect.width / 2);
    41→      const dy = (e.clientY - cy) / (rect.height / 2);
    42→      rotateX.set(-dy * 25);
    43→      rotateY.set(dx * 25);
    44→    };
    45→
    46→    const handleLeave = () => {
    47→      rotateX.set(0);
    48→      rotateY.set(0);
    49→    };
    50→
    51→    window.addEventListener('mousemove', handleMove);
    52→    window.addEventListener('mouseleave', handleLeave);
    53→    return () => {
    54→      window.removeEventListener('mousemove', handleMove);
    55→      window.removeEventListener('mouseleave', handleLeave);
    56→    };
    57→  }, [rotateX, rotateY]);
    58→
    59→  return (
    60→    <div ref={ref} className={`perspective-dramatic ${className}`}>
    61→      <motion.div
    62→        className="preserve-3d relative"
    63→        style={{ rotateX, rotateY, width: size, height: size }}
    64→      >
    65→        {/* Ball body */}
    66→        <div
    67→          className="w-full h-full rounded-full relative overflow-hidden"
    68→          style={{
    69→            boxShadow: `
    70→              0 0 ${size * 0.4}px rgba(192,57,43,0.18),
    71→              0 ${size * 0.12}px ${size * 0.35}px rgba(0,0,0,0.5),
    72→              0 0 ${size * 0.8}px rgba(192,57,43,0.08)
    73→            `,
    74→          }}
    75→        >
    76→          <img src="/cricket-ball.png" alt="" className="w-full h-full object-contain" />
    77→        </div>
    78→
    79→        {/* Shadow beneath ball */}
    80→        <div
    81→          className="absolute -bottom-6 left-1/2 -translate-x-1/2 w-3/4 h-4 rounded-full"
    82→          style={{
    83→            background: 'radial-gradient(ellipse, rgba(0,0,0,0.4) 0%, transparent 70%)',
    84→            filter: 'blur(8px)',
    85→          }}
    86→        />
    87→      </motion.div>
    88→    </div>
    89→  );
    90→}
