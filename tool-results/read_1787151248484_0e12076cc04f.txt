    37→    37→function loadSettings(): SettingsState {
    38→    38→  if (typeof window === 'undefined') {
    39→    39→    return {
    40→    40→      darkMode: true,
    41→    41→      language: 'English',
    42→    42→      emailNotifications: true,
    43→    43→      notifications: {
    44→    44→        matchStartAlerts: true,
    45→    45→        aiInsightDigest: true,
    46→    46→        playerPerformanceAlerts: false,
    47→    47→        weeklyAnalyticsReport: true,
    48→    48→      },
    49→    49→    };
    50→    50→  }
    51→    51→  try {
    52→    52→    const raw = localStorage.getItem('criciq-settings');
    53→    53→    if (raw) {
    54→    54→      const parsed = JSON.parse(raw) as Partial<SettingsState>;
    55→    55→      return {
    56→    56→        darkMode: parsed.darkMode ?? true,
    57→    57→        language: parsed.language ?? 'English',
    58→    58→        emailNotifications: parsed.emailNotifications ?? true,
    59→    59→        notifications: {
    60→    60→          matchStartAlerts: parsed.notifications?.matchStartAlerts ?? true,
    61→    61→          aiInsightDigest: parsed.notifications?.aiInsightDigest ?? true,
    62→    62→          playerPerformanceAlerts: parsed.notifications?.playerPerformanceAlerts ?? false,
    63→    63→          weeklyAnalyticsReport: parsed.notifications?.weeklyAnalyticsReport ?? true,
    64→    64→        },
    65→    65→      };
    66→    66→    }
    67→    67→  } catch {
    68→    68→    // ignore parse errors
    69→    69→  }
    70→    70→  return {
    71→    71→    darkMode: true,
    72→    72→    language: 'English',
    73→    73→    emailNotifications: true,
    74→    74→    notifications: {
    75→    75→      matchStartAlerts: true,
    76→    76→      aiInsightDigest: true,
    77→    77→      playerPerformanceAlerts: false,
    78→    78→      weeklyAnalyticsReport: true,
